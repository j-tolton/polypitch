cmake_minimum_required(VERSION 3.16)
project(polypitch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(EMSCRIPTEN)
  add_executable(polypitch dsp/polypitch.cpp dsp/wasm_api.cpp)
  target_compile_options(polypitch PRIVATE -O3)
  set_target_properties(polypitch PROPERTIES SUFFIX ".js")
  target_link_options(polypitch PRIVATE -O3 -sWASM=1 -sALLOW_MEMORY_GROWTH=1 -sMODULARIZE=1 -sEXPORT_NAME=createPolyPitch -sENVIRONMENT=web "-sEXPORTED_FUNCTIONS=_pp_init,_pp_input,_pp_analyse,_pp_lane_midi,_pp_lane_evidence,_pp_lane_pitch,_pp_lane_count,_pp_peak_midi,_pp_peak_strength,_pp_peak_count,_pp_noise_floor,_pp_frame_count,_pp_input_rms,_pp_spectral_centroid,_pp_background_mean,_pp_residual_mean,_pp_debug_top_hz,_pp_debug_top_db,_pp_debug_top_count" -sEXPORTED_RUNTIME_METHODS=HEAPF32)
else()
  add_executable(polypitch_evidence_test dsp/polypitch.cpp tests/evidence_test.cpp)
  target_include_directories(polypitch_evidence_test PRIVATE dsp)
  target_compile_options(polypitch_evidence_test PRIVATE -O2)
  enable_testing()
  add_test(NAME pitch_evidence COMMAND polypitch_evidence_test)
endif()
